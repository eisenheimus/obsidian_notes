
```bash 
# Устанавливаем fail2ban
sudo apt install fail2ban
```

```bash
# Создаем новый конфиг jail.conf
sudo touch /etc/fail2ban/jail.local
```

```ini
[DEFAULT]
# Список IP-адресов (или подсетей), которые НИКОГДА не будут заблокированы.
# 127.0.0.1 и ::1 — это localhost (сам сервер). Сюда же нужно добавить ваш белый IP. + Можно добавить второй IP, например белый.
ignoreip = 127.0.0.1 ::1 192.168.1.10
# Время, на которое блокируется нарушитель (1h = 1 час). 
# Можно писать в днях (d) или минутах (m).
bantime = 1h
# Окно времени, в течение которого считаются неудачные попытки (10m = 10 минут).
# Если за это время наберется maxretry ошибок, произойдет бан.
findtime = 10m
# Максимальное количество неудачных попыток входа за период findtime до блокировки.
maxretry = 3
# Начало секции настроек для защиты службы SSH.


[sshd]
# Включает (true) или выключает (false) защиту для этой конкретной службы.
enabled = true
# Указывает порт, на котором работает служба (ssh — это алиас для порта 22).
# ВНИМАНИЕ: Раз вы сменили порт SSH на 40175, эту строку лучше удалить, 
# чтобы Fail2Ban не путался и не пытался блокировать закрытый 22-й порт.
port = ssh
# Ваш новый кастомный порт SSH. Fail2Ban будет применять правила блокировки к этому порту.
port = + любой другой порт
# Путь к файлу лога, который Fail2Ban будет анализировать на предмет ошибок входа.
# Для Ubuntu это стандартный путь.
logpath = /var/log/auth.log
```

```
# Добавляем в автозапуск
sudo systemctl enable fail2ban

# Перезапускаем сервис fail2ban
sudo systemctl restart fail2ban
```

```bash
# Обзор общего состояния (какие правила защиты активны)
sudo fail2ban-client status

# Детальная проверка защиты SSH
sudo fail2ban-client status sshd

# Какие IP-адреса в данный момент заблокированы
sudo fail2ban-client banned

# Разблокировать IP-адрес 
sudo fail2ban-client set sshd unbanip 192.168.1.100
```
