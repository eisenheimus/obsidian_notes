```bash
# Удалите проблемный пакет, который мешает
sudo dpkg --remove --force-remove-reinstreq lxqt-panel

# Завершите все незавершённые настройки пакетов
sudo dpkg --configure -a

# Удалите полностью сломанные зависимости
sudo apt --fix-broken install -y

# Удалите ненужные старые ядра (ошибки из вашего лога)
sudo apt autoremove -y

# Очистите кэш пакетов
sudo apt clean
sudo apt autoclean
```
