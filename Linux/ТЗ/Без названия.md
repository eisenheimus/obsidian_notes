
  
## Решение задания 1

> Udev — определяет подключение USB-накопителя с нужным idVendor/idProduct и создаёт (или не создаёт) соответствующий объект в /dev/disk/by-uuid/ или устанавливает свойство ENV{UDISKS_AUTO}, влияющее на udisks. 
   Polkit — управляет правилами для udisks2 (стандартный механизм монтирования в современных Linux с GNOME/KDE/XFCE).
   Группы — GroupUSBNoPassword, GroupUSBPassword.
   Доступ к ttyACM* — через udev и проверку ID_SEAT (только физическая консоль).


 #### Bash скрипт
```bash
#!/bin/bash
# setup_usb_mount_policy.sh  

# Проверка root
if [ "$EUID" -ne 0 ]; then
    echo "Ошибка: запустите с sudo или от root" >&2
    exit 1
fi

set -e  

echo "Этап 1: Создание групп"
groupadd -f GroupUSBNoPassword
groupadd -f GroupUSBPassword

echo "Этап 2: Установка udisks2"
apt update && apt install -y udisks2 polkitd  

echo "Этап 3: Настройка udev"
cat > /etc/udev/rules.d/99-usb-mount-polkit.rules <<EOF
ACTION=="add", SUBSYSTEM=="block", ENV{ID_BUS}=="usb", \
ENV{ID_VENDOR_ID}=="0781", ENV{ID_MODEL_ID}=="5583", \
ENV{UDISKS_IGNORE}="0", ENV{UDISKS_SPECIAL}="1", TAG+="uaccess"
ACTION=="add", SUBSYSTEM=="block", ENV{ID_BUS}=="usb", \
ENV{UDISKS_IGNORE}="1"
EOF

echo "Этап 4: Настройка polkit"
cat > /etc/polkit-1/rules.d/99-usb-mount.rules <<EOF
polkit.addRule(function(action, subject) {
    if (action.id.indexOf("org.freedesktop.udisks2.filesystem-mount") === 0) {
        var special = action.lookup("special");
        if (special == "1") {
            if (subject.isInGroup("GroupUSBNoPassword")) {
                return polkit.Result.YES;
            }
            if (subject.isInGroup("GroupUSBPassword")) {
                return polkit.Result.AUTH_ADMIN_KEEP;
            }
            return polkit.Result.NO;
        } else {
            return polkit.Result.NO;
        }
    }
});
EOF

echo "Этап 5: Настройка прав доступа ttyACM*"
cat > /etc/udev/rules.d/98-ttyacm-local.rules <<EOF
SUBSYSTEM=="tty", KERNEL=="ttyACM[0-9]*", \
TAG+="seat", ENV{ID_SEAT}="seat0", TAG+="uaccess", \
OWNER="root", GROUP="dialout", MODE="0660"
EOF

echo "Этап 6: Настройка PAM access"
if ! grep -q "pam_access.so" /etc/pam.d/common-account 2>/dev/null; then
    echo "account required pam_access.so" >> /etc/pam.d/common-account
fi

cat > /etc/security/access.conf <<EOF
+:ALL:tty1 tty2 tty3 tty4 tty5 tty6
-:ALL:ALL
EOF

echo "Этап 7: Перезагрузка служб"
udevadm control --reload-rules
udevadm trigger
systemctl restart polkit  

echo "Скрипт завершен"
```


#### Проверка
```bash

#!/bin/bash
TEST_DEVICE="/dev/sdb1"

# Тест от пользователя без прав
sudo -u nobody udisksctl mount -b "$TEST_DEVICE" 2>&1 | grep -q "Not authorized" && \
    echo "Блокировка работает"
    
# Тест ttyACM
if echo "test" > /dev/ttyACM0 2>&1 | grep -q "Permission denied"; then
    echo "Доступ к ttyACM* запрещён (или нет устройства)"
fi

# Проверка прав
ls -la /dev/ttyACM0 2>/dev/null || echo "Устройство ttyACM0 не подключено"
```
